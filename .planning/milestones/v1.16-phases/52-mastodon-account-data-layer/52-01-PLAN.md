**File 1: Create `app/models/mastodon_account.rb`**

```ruby
class MastodonAccount < ApplicationRecord
  include Crud::ByUser

  belongs_to :user

  before_validation :parse_profile_url
  before_save :set_display_count

  validates :profile_url, presence: true
  validates :display_count, numericality: { greater_than: 0 }
  validate :profile_url_parseable

  scope :active, -> { where(deleted: false) }
  scope :recent, -> { order(created_at: :desc) }

  private

  def parse_profile_url
    return if profile_url.blank?
    uri = URI.parse(profile_url.strip)
    at_segment = uri.path.to_s.split('/').find { |p| p.start_with?('@') }
    self.instance = uri.host.to_s
    self.username = at_segment&.delete_prefix('@').to_s
  rescue URI::InvalidURIError
    self.instance = ''
    self.username = ''
  end

  def profile_url_parseable
    return if profile_url.blank?
    errors.add(:profile_url, :invalid) if instance.blank? || username.blank?
  end

  def set_display_count
    self.display_count = 5 if display_count.to_i == 0
  end
end
```

**Critical implementation notes:**
- Do NOT use `default_scope`. Use named `scope :active` — same as Note model.
- `parse_profile_url` runs as `before_validation`, so `profile_url_parseable` can check the parsed results.
- `URI.parse("not_a_url")` does NOT raise — it returns a URI::Generic with nil host. The code uses `uri.host.to_s` which gives `""` for nil, and `profile_url_parseable` catches the empty instance.
- `URI.parse("https://ruby.social/")` has no @-segment in path — `at_segment` is nil, `username` becomes `""`.
- Errors for unparseable URLs go on `:profile_url` (user-facing), not on `:instance` or `:username`.
- Do NOT add standalone `validates :instance, presence: true` or `validates :username, presence: true` — those would expose internal field names to the user.

require 'test_helper'

class MastodonAccountTest < ActiveSupport::TestCase

  def test_正常なURLを解析する
    account = MastodonAccount.new(user_id: 1, profile_url: 'https://ruby.social/@FastRuby')
    account.valid?
    assert_equal 'ruby.social', account.instance
    assert_equal 'FastRuby', account.username
  end

  def test_スキームなしURLは無効
    account = MastodonAccount.new(user_id: 1, profile_url: 'ruby.social/@FastRuby')
    assert_not account.valid?
    assert account.errors[:profile_url].any?, 'error should be on :profile_url'
    assert_empty account.errors[:instance], 'no error should be on :instance directly'
  end

  def test_パスにアットマークなしURLは無効
    account = MastodonAccount.new(user_id: 1, profile_url: 'https://ruby.social/')
    assert_not account.valid?
    assert account.errors[:profile_url].any?, 'error should be on :profile_url'
  end

  def test_不正なURLは無効
    account = MastodonAccount.new(user_id: 1, profile_url: 'not_a_url')
    assert_not account.valid?
    assert account.errors[:profile_url].any?, 'error should be on :profile_url'
  end

  def test_自分のアカウントは参照できる
    user = User.find(1)
    account = mastodon_accounts(:one)
    assert account.readable_by?(user)
    assert account.updatable_by?(user)
    assert account.deletable_by?(user)
  end

  def test_他人のアカウントは参照できない

    account = mastodon_accounts(:one)
    assert_not account.readable_by?(user)
    assert_not account.updatable_by?(user)
    assert_not account.deletable_by?(user)
  end

  def test_ソフトデリートで削除フラグが立つ
    account = mastodon_accounts(:one)
    account.destroy_logically!
    reloaded = MastodonAccount.find(account.id)
    assert reloaded.deleted, 'destroy_logically! should set deleted: true without removing the row'
  end

  def test_activeスコープは削除済みを除外する
    account = mastodon_accounts(:one)
    account.destroy_logically!
    assert_not_includes MastodonAccount.active, account
  end

end
