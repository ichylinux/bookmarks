```ruby
def test_twitter2_from_omniauth_stores_oauth2_tokens_on_existing_user
  u = users(:twitter_user)
  expires_ts = Time.now.to_i + 7200
  auth = OmniAuth::AuthHash.new(
    'provider' => 'twitter2',
    'uid' => u.uid,
    'info' => { 'name' => u.name },
    'credentials' => { 'token' => 'bearer-tok', 'refresh_token' => 'ref-tok', 'expires_at' => expires_ts, 'expires' => true }
  )
  result = User.from_omniauth(auth)
  assert_equal u.id, result.id
  u.reload
  assert_equal 'bearer-tok', u.oauth2_token
  assert_equal 'ref-tok', u.oauth2_refresh_token
  assert_in_delta expires_ts, u.oauth2_token_expires_at.to_i, 1
ensure
  u = users(:twitter_user)
  u.update_columns(oauth2_token: nil, oauth2_refresh_token: nil, oauth2_token_expires_at: nil)
end
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-90-01 | Tampering | `data['email']` in from_omniauth | mitigate | Email is written only when `data['email'].present?` AND user currently has a dummy email; a real email is never overwritten by OAuth input |
| T-90-02 | Spoofing | X-provided email value | accept | X is the authoritative identity provider for twitter2 users; trusting their email claim is consistent with trusting their UID claim (already accepted across all existing OAuth flows) |
| T-90-03 | Elevation of Privilege | Overwriting another user's email via collision | accept | `save(validate: false)` skips uniqueness check — same behavior as all other fields in this branch; single-user personal app, collision risk is negligible and accepted per CONTEXT.md |
| T-90-SC | Tampering | npm/pip/cargo installs | accept | No new packages installed in this phase |
