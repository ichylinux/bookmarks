```ruby
def parse_following_response(res)
  case res.status
  when 200
    body = parse_json_safe(res.body)
    return { success: false, error: :parse_error } unless body.is_a?(Hash)
    { success: true, payload: body }   # NOTE: returns :payload — the new parser returns :item
  when 401 then { success: false, error: :unauthorized }
  when 404 then { success: false, error: :not_found }
  when 429 then { success: false, error: :rate_limited }
  else { success: false, error: :api_error }
  end
end
```
